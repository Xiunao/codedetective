const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type, Authorization", "Access-Control-Max-Age": "86400" }; export default { async fetch(request, env, ctx) { // 1. 处理浏览器 OPTIONS 预检请求（解决 405 问题） if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders }); } // 2. 仅允许 POST if (request.method !== "POST") { return new Response(JSON.stringify({ error: "Method not allowed. Use POST." }), { status: 405, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } let body; try { body = await request.json(); } catch { return new Response(JSON.stringify({error:'Invalid JSON'}), {status:400, headers:corsHeaders}); } let code = body.code || ''; const apiKey = body.apiKey || ''; if (!code || !apiKey) { return new Response(JSON.stringify({error:'missing code or apiKey'}), {status:400, headers:corsHeaders}); } // 3. 长代码智能截断（防止超 500 字导致 API 失败或返回空） let queryStr = code; if (code.length > 500) { queryStr = code.slice(0, 500) + "...(代码过长已截断)"; } // 4. 调用腾讯云元宝 SearchPro（带 Cnt 参数，已验证可用） const r = await fetch('https://api.wsa.cloud.tencent.com/SearchPro', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + apiKey }, body: JSON.stringify({ Query: queryStr + ' 官方文档 API 是否存在 是否已弃用', Cnt: 10 }) }); const text = await r.text(); // 5. 501 兜底提示（方便前端排查 Key / 服务状态问题） if (r.status === 501) { return new Response(JSON.stringify({ error: "腾讯云 SearchPro 返回 501。请检查：1. Key 是否为腾讯云元宝搜索创建的 API Key；2. 服务是否已开通；3. 账号是否欠费。" }), { status: 501, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } // 6. 原样透传腾讯云响应 return new Response(text, { status: r.status, headers: { "Content-Type": "application/json", ...corsHeaders } }); } };

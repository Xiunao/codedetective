async function run(){ const code=document.getElementById('code').value.trim(); const key=document.getElementById('key').value.trim(); const btn=document.querySelector('button[onclick="run()"]'); if(!code){toast("先贴一段代码");return;} if(!key){ toast("未填 Key，已展示内置验真卡 ↓"); document.getElementById('cards').scrollIntoView({behavior:'smooth'}); return; } btn.disabled=true; btn.textContent="验真中…"; toast("正在通过中转调用元宝搜索…"); let live=document.getElementById('liveResult'); if(live) live.remove(); const controller=new AbortController(); const timer=setTimeout(()=>controller.abort(),15000); try{ const r=await fetch(DEFAULT_RELAY,{ method:"POST", headers:{"Content-Type":"application/json"}, body:JSON.stringify({code: code.length > 500 ? code.slice(0,500)+"...(已截断)" : code, apiKey:key}) const j=await r.json().catch(()=>null); if(!r.ok || !j){ let errMsg="请求失败"; if(r.status===401 || r.status===403) errMsg="Key 无效或无权限"; else if(r.status===501) errMsg="Worker 未更新，请确认已 Deploy"; else if(r.status===405) errMsg="请求方法被拒，请确认 Worker 已更新"; else if(j && j.error) errMsg=j.error; else errMsg="HTTP "+r.status; throw new Error(errMsg); } // 核心修复：解析 Pages（腾讯云返回的是 JSON 字符串数组） let pages=j?.Response?.Pages || []; pages = pages.map(p => { if(typeof p === 'string'){ try { return JSON.parse(p); } catch { return null; } } return p; }).filter(Boolean); if(!pages.length) throw new Error("元宝返回为空。请检查 Key 权限或代码是否有效"); const box=document.getElementById('cards'); live=document.createElement('div'); live.id='liveResult'; live.innerHTML=`
🔎 实时检索结果（元宝 SearchPro · 经 Cloudflare 中转）
真实调用成功
检索式
"${code.slice(0,80).replace(/"/g,'')}…"
检索来源
${pages.slice(0,5).map(p=>`
🌐 ${p.site || '未知站点'}${p.url?' · 打开原文':''}
${(p.passage||'').slice(0,350)}
`).join('')}
元宝搜索 APISearchPro${pages.length} 条来源
`; box.parentNode.insertBefore(live,box); toast("✅ 实时检索成功，共 "+pages.length+" 条来源"); live.scrollIntoView({behavior:'smooth'}); }catch(e){ clearTimeout(timer); const box=document.getElementById('cards'); live=document.createElement('div'); live.id='liveResult'; let hint=e.name==='AbortError'?"请求超时（15秒）":e.message; live.innerHTML=`
⚠️ 中转调用失败
回落到内置验真卡
${hint}
`; box.parentNode.insertBefore(live,box); }finally{ clearTimeout(timer); btn.disabled=false; btn.textContent="开始验真"; } }
